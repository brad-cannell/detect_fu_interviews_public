Identify Potential PHI Columns

These include (generally):

  • Columns with names
  • Columns with contact information (emails, phone numbers, etc.)
  • Columns with addresses
  • Columns with dates
  • Columns with information that can be linked to identify a single individual

Date Compliance

Ensure that temporal data complies with HIPAA Safe Harbor regulation

date_cols <- date_cols |>
    select(-PhoneBetterTime)

safe_harbor_deid <- function(df) {
  df |>
    mutate(
      # Parse dates (mm/dd/yyyy) and keep ONLY the year
      EntryDate              = year(mdy(EntryDate)),
      AppointmentDate        = year(mdy(AppointmentDate)),

      # Parse timestamps and keep ONLY the year
      xCreatedTimestamp      = year(mdy_hms(xCreatedTimestamp, tz = "UTC")),
      xModifiedTimestamp     = year(mdy_hms(xModifiedTimestamp, tz = "UTC")),

      # Remove fine-grained time information
      AppointmentTime        = NA,             # drop
      AppointmentTimeMilitary= NA,             # drop

      # Drop day names and day-of-month; keep only year-level fields
      cMocaOrientationDay       = NA,
      cMocaOrientationDayName   = NA,
      xRecordMonth              = NA,
      cMocaOrientationMonth     = NA,

      # Year fields are already OK
      xRecordYear            = as.integer(xRecordYear),
      cMocaOrientationYear   = as.integer(cMocaOrientationYear)
    )
}

clean_data <- safe_harbor_deid(clean_data)
